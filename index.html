Sistemdeki tüm otomatik bildirimlerin gerçek e-posta adreslerine ve WhatsApp numaralarına iletilmesi için ücretsiz servis bağlantı adreslerini tanımlayın.

1 Dakikada Ücretsiz Otomatik E-Posta + Google Drive Kalıcı Bulut Veritabanı Bağlantısı:

1. script.google.com adresine gidin ve Yeni Proje oluşturun.
2. Aşağıdaki kodu yapıştırıp sağ üstten Dağıt → Yeni Dağıtım → Web Uygulaması (Erişim: Herkes) seçerek dağıtın.
3. Size verilen https://script.google.com/macros/s/.../exec adresini aşağıdaki Otomatik E-Posta & Bulut Veritabanı URL kutusuna (veya Render.com CLOUD_DB_URL değişkenine) yapıştırın. Hem otomatik mailleriniz gönderilir hem de tüm verileriniz Google Drive'ınızda kalıcı olarak saklanır. 

var DB_FILE_NAME = "KaradenizECM_Bulut_Veritabani.json";

function doGet(e) {
  var files = DriveApp.getFilesByName(DB_FILE_NAME);
  if (files.hasNext()) {
    var content = files.next().getBlob().getDataAsString();
    return ContentService.createTextOutput(JSON.stringify({ ok: true, backup: JSON.parse(content) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: false })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  if (data.action === "SAVE_DB_BACKUP" && data.backup) {
    var files = DriveApp.getFilesByName(DB_FILE_NAME);
    var jsonStr = JSON.stringify(data.backup);
    if (files.hasNext()) { files.next().setContent(jsonStr); }
    else { DriveApp.createFile(DB_FILE_NAME, jsonStr, MimeType.PLAIN_TEXT); }
    return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
  }
  var mailOpts = {
    to: data.to || "",
    subject: data.subject || "Karadeniz ECM Bildirim",
    body: data.text || ""
  };
  if (data.cc) mailOpts.cc = data.cc;
  if (data.htmlBody) mailOpts.htmlBody = data.htmlBody;
  if (data.attachment && data.attachment.base64) {
    var bytes = Utilities.base64Decode(data.attachment.base64);
    var blob = Utilities.newBlob(bytes, data.attachment.mimeType, data.attachment.filename);
    mailOpts.attachments = [blob];
  }
  if (mailOpts.to) {
    MailApp.sendEmail(mailOpts);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}
